def test_pipeline_synthetic():
    """Test full pipeline on synthetic data."""
    print("\n" + "=" * 60)
    print("TEST: Full Pipeline (Synthetic)")
    print("=" * 60)
    
    n = 500
    np.random.seed(42)
    close = np.cumsum(np.random.normal(0, 5, n)) + 50000
    close = np.maximum(close, 1)
    
    dates = pd.date_range(start="2020-01-01", periods=n, freq="15min")
    df = pd.DataFrame({
        "timestamp": dates,
        "open": close * 0.999,
        "high": close + np.random.uniform(10, 50, n),
        "low": close - np.random.uniform(10, 50, n),
        "close": close,
        "volume": np.ones(n) * 100
    })
    
    research = RangeExpansionResearch()
    result = research.run_dataframe(df, dataset_id="synthetic_full_pipeline")
    
    checks = 0
    total = 11
    
    if result.status == "completed": checks += 1
    if result.statistics.get("true_range_status") == "complete": checks += 1
    if result.statistics.get("atr_status") == "complete": checks += 1
    if result.statistics.get("range_ratio_status") == "complete": checks += 1
    if result.statistics.get("percentiles_status") == "complete": checks += 1
    if result.statistics.get("states_status") == "complete": checks += 1
    if result.statistics.get("segments_status") == "complete": checks += 1
    if result.statistics.get("statistics_status") == "complete": checks += 1
    if result.statistics.get("mean_true_range", 0) > 0: checks += 1
    if result.statistics.get("mean_atr", 0) > 0: checks += 1
    if result.statistics.get("segment_count", 0) > 0: checks += 1
    
    labels = [
        "Completed", "TR", "ATR", "Ratio", "Percentiles",
        "States", "Segments", "Statistics",
        "Mean TR > 0", "Mean ATR > 0", "Segments > 0"
    ]
    
    for i, label in enumerate(labels):
        print(f"  {'✓' if i < checks else '✗'} {label}")
    
    print(f"\n  Pipeline: {checks}/{total} passed")
    return checks >= total - 1


def test_btc_data():
    """Test on all BTC datasets with correct statistics keys."""
    print("\n" + "=" * 60)
    print("TEST: BTC Data (All Timeframes)")
    print("=" * 60)
    
    datasets = {
        "BTC_15m": "data/raw/BTCUSD_15m.csv",
        "BTC_1h": "data/raw/BTCUSD_1h.csv",
        "BTC_4h": "data/raw/BTCUSD_4h.csv",
    }
    
    missing = [name for name, path in datasets.items() if not Path(path).exists()]
    if missing:
        print(f"  ⚠ Missing datasets: {missing}")
        return True
    
    registry = AtlasRegistry()
    runner = ModuleRunner(registry=registry)
    
    results = runner.run_on_datasets(
        module_class=RangeExpansionResearch,
        datasets=datasets,
        output_base_dir="outputs/module013",
        validate_after=False
    )
    
    all_ok = True
    for name, result in results.items():
        if result.status == "completed":
            # FIXED: Read correct statistics keys
            mean_tr = result.statistics.get("mean_true_range", 0)
            mean_atr = result.statistics.get("mean_atr", 0)
            mean_ratio = result.statistics.get("mean_ratio", 0)
            mean_pct = result.statistics.get("mean_percentile", 0)
            n_seg = result.statistics.get("segment_count", 0)
            n_bars = result.statistics.get("total_bars", 0)
            comp = result.statistics.get("compression_count", 0)
            norm = result.statistics.get("normal_count", 0)
            expn = result.statistics.get("expansion_count", 0)
            extr = result.statistics.get("extreme_count", 0)
            
            print(f"\n  {name}:")
            print(f"    Mean TR:        {mean_tr:,.4f}")
            print(f"    Mean ATR:       {mean_atr:,.4f}")
            print(f"    Mean Ratio:     {mean_ratio:.4f}")
            print(f"    Mean Percentile: {mean_pct:.2f}")
            print(f"    Segments:       {n_seg:,}")
            print(f"    Total bars:     {n_bars:,}")
            print(f"    States: C={comp:,} N={norm:,} E={expn:,} X={extr:,}")
            
            # Verify artifacts exist
            expected_artifacts = [
                'true_range', 'atr', 'range_ratio', 'range_percentile',
                'expansion_states', 'segments', 'behavior_statistics',
                'transition_counts', 'transition_probabilities',
                'duration_distribution', 'occupancy_statistics',
                'persistence_statistics'
            ]
            present = [a for a in expected_artifacts if a in result.artifacts]
            print(f"    Artifacts: {len(present)}/{len(expected_artifacts)} exported")
            
            if len(present) < len(expected_artifacts):
                missing_artifacts = set(expected_artifacts) - set(present)
                print(f"    ⚠ Missing artifacts: {missing_artifacts}")
                all_ok = False
        else:
            print(f"  ✗ {name}: Failed - {result.errors}")
            all_ok = False
    
    return all_ok
