# atlas/modules/module015_compression/module.py
"""
Module 015: Compression Analysis - Phase 1

Analyzes compression behavior after range contraction.
Reuses atlas.core.volatility for percentile calculations.
Uses atlas.core.compression for detection and statistics.

FIXED: Config-driven ATR and percentile windows. Duration distribution exported.
       Logger improved. Version bumped to 1.0.0.
"""

from typing import Dict, Any, Optional, List
from pathlib import Path

import numpy as np
import pandas as pd

from atlas.core.base_research_module import BaseResearchModule
from atlas.core.volatility import RangeMetrics
from atlas.core.compression import CompressionDetector, CompressionStatistics
from atlas.framework.registry import AtlasRegistry

from .config import (
    MODULE_ID, MODULE_NAME, MODULE_VERSION,
    ATR_WINDOW, PERCENTILE_WINDOW,
    COMPRESSION_THRESHOLD, POST_COMPRESSION_LOOKAHEAD, MIN_SEGMENT_LENGTH,
)
from .protocol import PROTOCOL

MODULE_INFO = {
    "id": MODULE_ID, "name": MODULE_NAME, "version": MODULE_VERSION,
    "author": "Project Atlas",
    "phase": "Phase 1 - Compression Analysis",
    "requires": ["000", "013"],
    "datasets_required": ["BTC_15m", "BTC_1h", "BTC_4h"],
    "produces": ["compression_state", "compression_segments",
                 "compression_statistics", "post_compression_moves",
                 "compression_duration_distribution"],
    "status": "Stable - 1.0.0",
    "core_utils": "atlas.core.compression, atlas.core.volatility",
}


class CompressionResearch(BaseResearchModule):
    """Module 015: Compression Analysis - Phase 1"""
    
    PROTOCOL = PROTOCOL
    MODULE_INFO = MODULE_INFO
    
    def __init__(self, registry: Optional[AtlasRegistry] = None):
        super().__init__(module_name=MODULE_NAME, module_version=MODULE_VERSION)
        self.registry = registry or AtlasRegistry()
        existing = self.registry.get_module(MODULE_ID)
        if not existing:
            self.registry.register_module(self.PROTOCOL)
        self.range_ratio: Optional[pd.Series] = None
        self.range_percentile: Optional[pd.Series] = None
        self.compression: Optional[pd.Series] = None
        self.segments: Optional[pd.DataFrame] = None
        self.post_moves: Optional[pd.DataFrame] = None
    
    def _register_pipeline_steps(self):
        self.pipeline_steps = [
            ("range_percentiles", self._compute_range_percentiles),
            ("compression_state", self._identify_compression),
            ("compression_segments", self._detect_segments),
            ("duration_statistics", self._compute_duration_stats),
            ("post_compression", self._measure_post_compression),
            ("summary", self._compute_summary),
            ("export", self._export_results),
        ]
    
    @property
    def pipeline_names(self) -> List[str]:
        return [name for name, _ in self.pipeline_steps]
    
    def get_config_snapshot(self) -> Dict[str, Any]:
        return {
            "module_name": self.module_name, "module_version": self.module_version,
            "atr_window": ATR_WINDOW, "percentile_window": PERCENTILE_WINDOW,
            "compression_threshold": COMPRESSION_THRESHOLD,
            "post_compression_lookahead": POST_COMPRESSION_LOOKAHEAD,
            "min_segment_length": MIN_SEGMENT_LENGTH,
        }
    
    def get_protocol_snapshot(self) -> Dict[str, Any]:
        return {
            "question": PROTOCOL.question, "hypothesis": PROTOCOL.hypothesis,
            "methodology": PROTOCOL.methodology, "depends_on": PROTOCOL.depends_on,
        }
    
    # ---- Step 1: Compute Range Percentiles ----
    def _compute_range_percentiles(self) -> Dict[str, Any]:
        """Compute range percentiles from OHLCV data using shared core."""
        self.logger.info("Step 1: Computing range percentiles")
        self.logger.info(f"  Using ATR window={ATR_WINDOW}, Percentile window={PERCENTILE_WINDOW}")
        
        self.range_ratio = RangeMetrics.range_ratio(self.df, ATR_WINDOW)
        self.range_percentile = RangeMetrics.rolling_percentile(self.range_ratio, PERCENTILE_WINDOW)
        self.df['range_percentile'] = self.range_percentile
        
        n = len(self.range_percentile)
        warmup = int(self.range_percentile.isna().sum())
        valid = int(self.range_percentile.notna().sum())
        
        self.logger.metric("Percentile observations", f"{n:,}")
        self.logger.metric("Percentile warm-up NaN", f"{warmup}")
        self.logger.metric("Valid percentiles", f"{valid:,}")
        
        return {
            "percentile_observations": n,
            "percentile_warmup_nan": warmup,
            "valid_percentiles": valid,
        }
    
    # ---- Step 2: Identify Compression ----
    def _identify_compression(self) -> Dict[str, Any]:
        """Identify bars in compression state."""
        self.logger.info("Step 2: Identifying compression states")
        self.logger.info(f"  Threshold: percentile < {COMPRESSION_THRESHOLD}")
        
        self.compression = CompressionDetector.identify_compression(
            self.range_percentile, COMPRESSION_THRESHOLD
        )
        self.df['is_compression'] = self.compression
        
        n_compression = int(self.compression.sum())
        n_total = len(self.compression)
        
        self.logger.metric("Total bars", f"{n_total:,}")
        self.logger.metric("Compression bars", f"{n_compression:,} ({n_compression/n_total*100:.1f}%)")
        
        self.result.artifacts['compression_state'] = self.df[['timestamp', 'range_percentile', 'is_compression']].copy()
        
        return {
            "total_bars": n_total,
            "compression_bars": n_compression,
            "compression_pct": round(n_compression / n_total * 100, 1) if n_total > 0 else 0,
        }
    
    # ---- Step 3: Detect Segments ----
    def _detect_segments(self) -> Dict[str, Any]:
        """Detect contiguous compression segments."""
        self.logger.info("Step 3: Detecting compression segments")
        
        self.segments = CompressionDetector.detect_compression_segments(self.compression)
        
        n_seg = len(self.segments)
        
        self.logger.metric("Compression segments", f"{n_seg:,}")
        if n_seg > 0:
            self.logger.metric("Longest segment", f"{self.segments['duration_bars'].max():,} bars")
            self.logger.metric("Mean length", f"{self.segments['duration_bars'].mean():.1f} bars")
        
        self.result.artifacts['compression_segments'] = self.segments.copy()
        
        return {"compression_segments": n_seg}
    
    # ---- Step 4: Duration Statistics ----
    def _compute_duration_stats(self) -> Dict[str, Any]:
        """Compute compression duration statistics."""
        self.logger.info("Step 4: Computing duration statistics")
        
        stats = CompressionStatistics.duration_statistics(self.segments)
        
        if stats['segment_count'] > 0:
            self.logger.info(f"  Segments: {stats['segment_count']:,}")
            self.logger.info(f"  Mean duration: {stats['mean_duration']:.1f} bars")
            self.logger.info(f"  Median duration: {stats['median_duration']:.0f} bars")
            self.logger.info(f"  Max duration: {stats['max_duration']:,} bars")
            self.logger.info(f"  P95 duration: {stats['p95_duration']:.0f} bars")
        
        # Duration distribution for export
        if len(self.segments) > 0:
            duration_distribution = (
                self.segments
                .groupby("duration_bars")
                .size()
                .reset_index(name="count")
            )
            self.result.artifacts["compression_duration_distribution"] = duration_distribution
        
        return {
            "segment_count": stats['segment_count'],
            "mean_duration": stats['mean_duration'],
            "median_duration": stats['median_duration'],
            "std_duration": stats['std_duration'],
            "min_duration": stats['min_duration'],
            "max_duration": stats['max_duration'],
            "p95_duration": stats['p95_duration'],
            "p99_duration": stats['p99_duration'],
            "total_compression_bars": stats['total_bars'],
        }
    
    # ---- Step 5: Post-Compression Movement ----
    def _measure_post_compression(self) -> Dict[str, Any]:
        """Measure price movement after compression ends."""
        self.logger.info("Step 5: Measuring post-compression movement")
        self.logger.info(f"  Lookahead: {POST_COMPRESSION_LOOKAHEAD} bars")
        
        close_prices = self.df['close']
        
        self.post_moves = CompressionDetector.measure_post_compression(
            self.segments, close_prices, POST_COMPRESSION_LOOKAHEAD
        )
        
        n_moves = len(self.post_moves)
        
        if n_moves > 0:
            for bars in POST_COMPRESSION_LOOKAHEAD:
                col = f'pct_move_{bars}b'
                if col in self.post_moves.columns:
                    mean_move = self.post_moves[col].mean()
                    up_pct = (self.post_moves[col] > 0).mean() * 100
                    self.logger.metric(f"  {bars}b post-compression", f"mean={mean_move:+.4f}%, up={up_pct:.0f}%")
        
        self.result.artifacts['post_compression_moves'] = self.post_moves.copy()
        
        return {"post_compression_events": n_moves}
    
    # ---- Step 6: Summary ----
    def _compute_summary(self) -> Dict[str, Any]:
        """Compute overall summary statistics with flat keys."""
        self.logger.info("Step 6: Computing summary statistics")
        
        freq = CompressionStatistics.compression_frequency(self.compression)
        post_summary = CompressionStatistics.post_compression_summary(self.post_moves)
        
        summary = {}
        summary.update(freq)
        summary.update(post_summary)
        
        for key, value in summary.items():
            self.result.statistics[key] = value
        
        summary_df = pd.DataFrame({"metric": list(summary.keys()), "value": list(summary.values())})
        self.result.artifacts['compression_statistics'] = summary_df
        
        self.logger.success("Summary statistics complete")
        
        return {"summary_status": "complete"}
    
    # ---- Step 7: Export ----
    def _export_results(self) -> Dict[str, Any]:
        """Export all results."""
        self.logger.info("Step 7: Exporting results")
        self.logger.success("Export complete")
        return {"export_status": "complete"}
    
    def compute(self) -> Dict[str, Any]:
        self.logger.section("Executing Pipeline (Phase 1)")
        results = self.execute_pipeline()
        self.result.status = "completed"
        self.logger.success("Module 015 Phase 1 complete")
        return results
    
    def run(self, filepath: str, output_dir: Optional[str] = None):
        return super().run(filepath, output_dir)
    
    def run_dataframe(self, df: pd.DataFrame, dataset_id: str = "in_memory",
                      output_dir: Optional[str] = None):
        return super().run_dataframe(df, dataset_id, output_dir)
