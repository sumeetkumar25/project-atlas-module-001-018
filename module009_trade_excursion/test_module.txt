def create_price_dataframe(prices: np.ndarray) -> pd.DataFrame:
    """Create OHLCV DataFrame from price series"""
    n = len(prices)
    dates = pd.date_range(start="2020-01-01", periods=n, freq="15min")
    
    return pd.DataFrame({
        "timestamp": dates,
        "open": prices,
        "high": prices * 1.005,
        "low": prices * 0.995,
        "close": prices,
        "volume": np.ones(n) * 100
    })


def test_synthetic_perfect_continuation():
    """
    Test 1: Price only goes up after entry.
    
    Entry at 100, future: 102, 104, 106, 108
    MFE = 8%, MAE = 0%
    """
    print("\n" + "=" * 60)
    print("SYNTHETIC TEST 1: Perfect Continuation")
    print("=" * 60)
    
    # Create synthetic events manually (bypassing chain)
    events = pd.DataFrame([{
        "trend_id": 1,
        "trend_direction": "Up",
        "pullback_depth_pct": 2.0,
        "pullback_duration": 5,
        "pullback_end_price": 100.0,
        "pullback_end_idx": 0,
        "outcome": "continuation"
    }])
    
    # Price only goes up
    prices = np.array([100, 102, 104, 106, 108])
    df = create_price_dataframe(prices)
    
    research = TradeExcursionResearch()
    research.set_events(events)
    result = research.run_dataframe(df, dataset_id="synthetic_continuation")
    
    mfe = result.statistics.get("mean_mfe", 0)
    mae = result.statistics.get("mean_mae", 0)
    rr = result.statistics.get("mean_rr", 0)
    
    print(f"\n  Prices after entry: {prices[1:].tolist()}")
    print(f"  MFE: {mfe:.2f}% (expected: 8%)")
    print(f"  MAE: {mae:.2f}% (expected: 0%)")
    print(f"  RR: {rr:.2f}")
    
    checks_passed = 0
    
    # MFE should be ~8%
    if abs(mfe - 8.0) < 0.5:
        print(f"  ✓ MFE correct ({mfe:.1f}%)")
        checks_passed += 1
    else:
        print(f"  ✗ MFE wrong: {mfe:.1f}% (expected 8%)")
    
    # MAE should be 0%
    if mae == 0:
        print(f"  ✓ MAE = 0% (no drawdown)")
        checks_passed += 1
    else:
        print(f"  ✗ MAE should be 0%, got {mae:.2f}%")
    
    # MFE >= 0, MAE >= 0
    if mfe >= 0 and mae >= 0:
        print(f"  ✓ Both MFE and MAE non-negative")
        checks_passed += 1
    
    print(f"\n  Test: {checks_passed}/3 passed")
    return checks_passed == 3

def test_compute_mfe_mae_function():
    """Test the core compute_mfe_mae function directly"""
    print("\n" + "=" * 60)
    print("UNIT TEST: compute_mfe_mae()")
    print("=" * 60)
    
    checks_passed = 0
    
    # Test 1: Uptrend, prices go up
    mfe, mae, _, _, _, _ = compute_mfe_mae(100, np.array([102, 104, 106]), "Up")
    if abs(mfe - 6.0) < 0.01 and mae == 0:
        print(f"  ✓ Uptrend up: MFE=6%, MAE=0%")
        checks_passed += 1
    
    # Test 2: Uptrend, prices go down
    mfe, mae, _, _, _, _ = compute_mfe_mae(100, np.array([98, 96, 94]), "Up")
    if mfe == 0 and abs(mae - 6.0) < 0.01:
        print(f"  ✓ Uptrend down: MFE=0%, MAE=6%")
        checks_passed += 1
    
    # Test 3: Downtrend, prices go down
    mfe, mae, _, _, _, _ = compute_mfe_mae(100, np.array([98, 96, 94]), "Down")
    if abs(mfe - 6.0) < 0.01 and mae == 0:
        print(f"  ✓ Downtrend down: MFE=6%, MAE=0%")
        checks_passed += 1
    
    # Test 4: Downtrend, prices go up
    mfe, mae, _, _, _, _ = compute_mfe_mae(100, np.array([102, 104, 106]), "Down")
    if mfe == 0 and abs(mae - 6.0) < 0.01:
        print(f"  ✓ Downtrend up: MFE=0%, MAE=6%")
        checks_passed += 1
    
    # Test 5: Mixed
    mfe, mae, _, _, _, _ = compute_mfe_mae(100, np.array([103, 98, 106]), "Up")
    if abs(mfe - 6.0) < 0.01 and abs(mae - 2.0) < 0.01:
        print(f"  ✓ Mixed: MFE=6%, MAE=2%")
        checks_passed += 1
    
    # Test 6: MFE >= 0, MAE >= 0
    mfe, mae, _, _, _, _ = compute_mfe_mae(100, np.array([100, 100, 100]), "Up")
    if mfe >= 0 and mae >= 0:
        print(f"  ✓ Flat: MFE={mfe}%, MAE={mae}% (both ≥ 0)")
        checks_passed += 1
    
    print(f"\n  Unit Tests: {checks_passed}/6 passed")
    return checks_passed == 6


def test_btc_datasets():
    """Test on BTC real data via full chain"""
    print("\n" + "=" * 60)
    print("BTC DATASETS (Full Chain: 005→007→008→009)")
    print("=" * 60)
    
    registry = AtlasRegistry()
    runner = ModuleRunner(registry=registry)
    
    datasets = {
        "BTC_15m": "data/raw/BTCUSD_15m.csv",
        "BTC_1h": "data/raw/BTCUSD_1h.csv",
        "BTC_4h": "data/raw/BTCUSD_4h.csv",
    }
    
    missing = [name for name, path in datasets.items() if not Path(path).exists()]
    if missing:
        print(f"\n✗ Missing: {missing}")
        return False
    
    results = runner.run_on_datasets(
        module_class=TradeExcursionResearch,
        datasets=datasets,
        output_base_dir="outputs/module009",
        validate_after=False
    )
    
    all_passed = True
    
    for name, result in results.items():
        if result.status == "completed":
            n = result.statistics.get("events_analyzed", 0)
            mfe = result.statistics.get("mean_mfe", 0)
            mae = result.statistics.get("mean_mae", 0)
            rr = result.statistics.get("mean_rr", 0)
            p_mfe_5 = result.statistics.get("p_mfe_gt_5", 0)
            
            print(f"\n  {name}:")
            print(f"    Events: {n:,}")
            print(f"    Mean MFE: {mfe:.2f}%")
            print(f"    Mean MAE: {mae:.2f}%")
            print(f"    Mean RR: {rr:.2f}")
            print(f"    P(MFE > 5%): {p_mfe_5}%")
            
            if n > 0:
                print(f"    ✓ Analysis complete")
            else:
                print(f"    ⚠ No events")
        else:
            print(f"  ✗ {name}: Failed")
            all_passed = False
    
    return all_passed
