def create_price_dataframe(prices: np.ndarray) -> pd.DataFrame:
    """Create OHLCV DataFrame from price series"""
    n = len(prices)
    dates = pd.date_range(start="2020-01-01", periods=n, freq="15min")
    
    return pd.DataFrame({
        "timestamp": dates,
        "open": prices,
        "high": prices * 1.001,
        "low": prices * 0.999,
        "close": prices,
        "volume": np.ones(n) * 100
    })


def get_trends_from_module005(df: pd.DataFrame) -> pd.DataFrame:
    """
    Run Module 005 on the DataFrame and return trend segments.
    
    This is the CORRECT way to get trends for Module 007.
    Module 007 should never define its own trends.
    """
    trend_module = TrendPersistenceResearch()
    trend_module.MIN_TREND_LENGTH = 1  # Get all trends for testing
    
    result = trend_module.run_dataframe(df, dataset_id="test_trends")
    
    if 'trend_segments' in result.artifacts:
        trends = result.artifacts['trend_segments']
        print(f"  Module 005 found {len(trends)} trend segments:")
        for _, t in trends.iterrows():
            print(f"    Trend {int(t['segment_id'])}: {t['direction_label']} "
                  f"[{int(t['start_idx'])}→{int(t['end_idx'])}] length={int(t['length'])}")
        return trends
    else:
        print(f"  Module 005 found NO trend segments")
        return pd.DataFrame()


def test_synthetic_staircase():
    """
    Test 1: Known staircase pattern.
    
    First run Module 005 to see what trends it identifies.
    Then run Module 007 on those trends.
    
    Prices: 100 → 110 → 120 → 115 → 125
    """
    print("\n" + "=" * 60)
    print("SYNTHETIC TEST 1: Staircase Pattern")
    print("=" * 60)
    
    prices = np.array([100, 110, 120, 115, 125])
    df = create_price_dataframe(prices)
    
    print(f"\n  Prices: {prices.tolist()}")
    
    # Step 1: Get trends from Module 005
    trends = get_trends_from_module005(df)
    
    if len(trends) == 0:
        print(f"  ⚠ Module 005 found no trends - cannot test pullbacks")
        print(f"  This is expected for very short price series")
        return True  # Not a failure of Module 007
    
    # Step 2: Run Module 007 with those trends
    research = PullbackResearch()
    research.set_trend_segments(trends)
    result = research.run_dataframe(df, dataset_id="synthetic_staircase")
    
    n_pullbacks = result.statistics.get("pullback_count", 0)
    
    print(f"\n  Module 007 results:")
    print(f"  Pullbacks found: {n_pullbacks}")
    
    if n_pullbacks > 0:
        mean_depth = result.statistics.get("mean_depth_pct", 0)
        recovery_rate = result.statistics.get("recovery_rate", 0)
        
        print(f"  Mean depth: {mean_depth:.2f}%")
        print(f"  Recovery rate: {recovery_rate*100:.0f}%")
        
        # Show each pullback
        if 'pullbacks' in result.artifacts:
            for _, pb in result.artifacts['pullbacks'].iterrows():
                print(f"  Pullback after trend {int(pb['trend_id'])}: "
                      f"depth={pb['pullback_depth_pct']:.2f}%, "
                      f"duration={int(pb['pullback_duration'])} bars, "
                      f"recovered={pb['recovered']}")
    
    print(f"\n  ✓ Test complete (pullbacks depend on Module 005 output)")
    return True

def test_synthetic_flat():
    """
    Test 3: Flat market → no trends → no pullbacks.
    """
    print("\n" + "=" * 60)
    print("SYNTHETIC TEST 3: Flat Market")
    print("=" * 60)
    
    prices = np.ones(20) * 100
    df = create_price_dataframe(prices)
    
    # Step 1: Get trends from Module 005
    trends = get_trends_from_module005(df)
    
    # Step 2: Run Module 007
    research = PullbackResearch()
    if len(trends) > 0:
        research.set_trend_segments(trends)
    
    result = research.run_dataframe(df, dataset_id="synthetic_flat")
    
    n_pullbacks = result.statistics.get("pullback_count", 0)
    
    print(f"\n  Module 007 results:")
    print(f"  Pullbacks found: {n_pullbacks}")
    
    if n_pullbacks == 0:
        print(f"  ✓ Correctly found no pullbacks in flat market")
        return True
    else:
        print(f"  ✗ Expected 0 pullbacks, got {n_pullbacks}")
        return False


def test_btc_datasets():
    """Test 4: BTC real data"""
    print("\n" + "=" * 60)
    print("BTC DATASETS")
    print("=" * 60)
    
    registry = AtlasRegistry()
    runner = ModuleRunner(registry=registry)
    
    datasets = {
        "BTC_15m": "data/raw/BTCUSD_15m.csv",
        "BTC_1h": "data/raw/BTCUSD_1h.csv",
        "BTC_4h": "data/raw/BTCUSD_4h.csv",
    }
    
    missing = [name for name, path in datasets.items() if not Path(path).exists()]
    if missing:
        print(f"\n✗ Missing: {missing}")
        return False
    
    results = runner.run_on_datasets(
        module_class=PullbackResearch,
        datasets=datasets,
        output_base_dir="outputs/module007",
        validate_after=False
    )
    
    all_passed = True
    
    for name, result in results.items():
        if result.status == "completed":
            n_pullbacks = result.statistics.get("pullback_count", 0)
            mean_depth = result.statistics.get("mean_depth_pct", 0)
            median_depth = result.statistics.get("median_depth_pct", 0)
            mean_dur = result.statistics.get("mean_duration", 0)
            recovery = result.statistics.get("recovery_rate", 0)
            
            print(f"\n  {name}:")
            print(f"    Pullbacks: {n_pullbacks:,}")
            print(f"    Mean depth: {mean_depth:.2f}%")
            print(f"    Median depth: {median_depth:.2f}%")
            print(f"    Mean duration: {mean_dur:.1f} bars")
            print(f"    Recovery rate: {recovery*100:.1f}%")
            
            if n_pullbacks > 0:
                print(f"    ✓ Pullback analysis complete")
            else:
                print(f"    ⚠ No pullbacks found")
        else:
            print(f"  ✗ {name}: Failed")
            all_passed = False
    
    return all_passed
