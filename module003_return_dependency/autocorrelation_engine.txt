"""
Autocorrelation Engine - Phase 1

Measures return autocorrelation for lags 1 through N.

Statistical notes:
- Currently uses ±1.96/sqrt(n) confidence bounds (Bartlett's approximation)
- This is the standard first-order approximation

TODO (Phase 2):
- Replace fixed confidence bounds with Bartlett-adjusted intervals
  Var[r_k] ≈ (1/n) * (1 + 2*sum_{j=1}^{k-1} r_j²)
  This accounts for cumulative variance at higher lags.

TODO (Phase 2):
- Replace simple "count > 2*expected" significance test with:
  - Binomial test: P(X ≥ k) where X ~ Binomial(100, 0.05)
  - Or Poisson approximation: P(X ≥ k) where λ = 5
  This provides proper p-values instead of binary threshold.
"""

import numpy as np
import pandas as pd
from typing import Optional
from scipy import stats as scipy_stats


class AutocorrelationEngine:
    """
    Computes autocorrelation function (ACF) for return series.
    
    TODO (Phase 2): Bartlett-adjusted confidence intervals
    TODO (Phase 2): Binomial/Poisson test for significance count
    """
    
    def __init__(self, max_lag: int = 100):
        self.max_lag = max_lag
        self._confidence_bounds: Optional[dict] = None
    
    def compute(self, returns: pd.Series) -> pd.DataFrame:
        """Compute autocorrelation for lags 1 through max_lag"""
        clean = returns.dropna()
        n = len(clean)
        
        if n < self.max_lag + 1:
            raise ValueError(f"Need at least {self.max_lag + 1} observations, got {n}")
        
        # Standard error under H0 (Bartlett's approximation)
        # TODO (Phase 2): Use Bartlett's formula with cumulative adjustment
        # Var[r_k] ≈ (1/n) * (1 + 2*sum_{j=1}^{k-1} r_j²)
        se = 1.0 / np.sqrt(n)
        
        lower_bound = -1.96 * se
        upper_bound = 1.96 * se
        
        self._confidence_bounds = {
            'n_observations': n,
            'standard_error': se,
            'lower_95ci': lower_bound,
            'upper_95ci': upper_bound,
            'method': 'Bartlett approximation (fixed)',
            'todo': 'Replace with Bartlett cumulative adjustment in Phase 2'
        }
        
        acf_values = []
        for lag in range(1, self.max_lag + 1):
            value = clean.autocorr(lag=lag)
            acf_values.append({
                "lag": lag,
                "autocorrelation": value,
                "abs_autocorrelation": abs(value),
                "significant": abs(value) > upper_bound,
                "lower_ci": lower_bound,
                "upper_ci": upper_bound
            })
        
        return pd.DataFrame(acf_values)
    
    def get_confidence_bounds(self) -> dict:
        """Get confidence interval bounds"""
        return self._confidence_bounds or {}
    
    def get_significant_lags(self, acf_df: pd.DataFrame) -> pd.DataFrame:
        """Extract significant lags"""
        return acf_df[acf_df['significant']].copy()
    
    def get_strongest_lags(self, acf_df: pd.DataFrame, n: int = 5) -> pd.DataFrame:
        """Get n strongest lags"""
        return acf_df.nlargest(n, 'abs_autocorrelation')
    
    def binomial_test(self, acf_df: pd.DataFrame) -> dict:
        """
        Binomial test for number of significant lags.
        
        TODO (Phase 2): Make this the primary significance test.
        Currently implemented but results are advisory only.
        
        H0: Each lag has 5% chance of being significant by chance
        Test: P(X ≥ k) where X ~ Binomial(n_lags, 0.05)
        """
        n_lags = len(acf_df)
        n_significant = acf_df['significant'].sum()
        
        # P(X ≥ k) = 1 - P(X ≤ k-1)
        p_value = 1.0 - scipy_stats.binom.cdf(n_significant - 1, n_lags, 0.05)
        
        return {
            'n_lags': n_lags,
            'n_significant': int(n_significant),
            'expected': n_lags * 0.05,
            'p_value': float(p_value),
            'significant_at_5pct': p_value < 0.05,
            'method': 'Binomial test',
            'note': 'Phase 2 will use this as primary test'
        }
    
    def summary(self, acf_df: pd.DataFrame) -> str:
        """Generate ACF summary"""
        bounds = self._confidence_bounds or {}
        n = bounds.get('n_observations', 0)
        se = bounds.get('standard_error', 0)
        
        lines = []
        lines.append(f"Autocorrelation Analysis (n={n:,} observations)")
        lines.append(f"Standard Error: {se:.6f} (Bartlett approximation)")
        lines.append(f"95% CI: [{bounds.get('lower_95ci', 0):.6f}, {bounds.get('upper_95ci', 0):.6f}]")
        lines.append(f"Method: {bounds.get('method', 'N/A')}")
        lines.append("")
        
        # Significant lags (current method)
        significant = self.get_significant_lags(acf_df)
        n_significant = len(significant)
        expected = self.max_lag * 0.05
        
        lines.append(f"Significant lags: {n_significant}/{self.max_lag}")
        lines.append(f"Expected under H0: ~{expected:.1f}")
        
        # Binomial test (advisory)
        binom = self.binomial_test(acf_df)
        lines.append(f"Binomial p-value: {binom['p_value']:.4f}")
        
        if binom['significant_at_5pct']:
            lines.append("⚠ Binomial test: significant excess (p < 0.05)")
            lines.append("   Evidence AGAINST return independence")
        else:
            lines.append("✓ Binomial test: not significant (p ≥ 0.05)")
        
        # TODO note
        lines.append("")
        lines.append("NOTE: Phase 2 will use Bartlett-adjusted CIs and")
        lines.append("      binomial test as primary significance method.")
        
        # Strongest lags
        lines.append("")
        lines.append("Strongest autocorrelations:")
        strongest = self.get_strongest_lags(acf_df, n=5)
        for _, row in strongest.iterrows():
            sig = " *" if row['significant'] else ""
            lines.append(f"  Lag {int(row['lag']):3d}: {row['autocorrelation']:+.6f}{sig}")
        
        return "\n".join(lines)
