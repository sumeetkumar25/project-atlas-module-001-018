import sys

def test_import():

    module = VolumeBehaviourResearch()

    assert module.module_name == "Volume Behaviour"

    print("✓ Import passed")

    return True


def test_pipeline():

    print("\n" + "=" * 60)
    print("SYNTHETIC PIPELINE TEST")
    print("=" * 60)

    n = 500
    np.random.seed(42)
    dates = pd.date_range(start="2020-01-01", periods=n, freq="15min")

    df = pd.DataFrame({
        "timestamp": dates,
        "open": np.random.uniform(50000, 51000, n),
        "high": np.random.uniform(50100, 51100, n),
        "low": np.random.uniform(49900, 50900, n),
        "close": np.random.uniform(50000, 51000, n),
        "volume": np.random.uniform(10, 500, n),
    })

    research = VolumeBehaviourResearch()
    result = research.run_dataframe(df, dataset_id="pipeline_test")

    stats = result.statistics
    checks = 0

    labels = [
        "Completed",
        "Clean", "Normalize", "Percentile", "State",
        "Segments", "Statistics", "Export",
    ]

    total = len(labels)

    if result.status == "completed":
        checks += 1
    if stats.get("clean_status") == "complete":
        checks += 1
    if stats.get("normalization_status") == "complete":
        checks += 1
    if stats.get("percentile_status") == "complete":
        checks += 1
    if stats.get("state_status") == "complete":
        checks += 1
    if stats.get("segments_status") == "complete":
        checks += 1
    if stats.get("statistics_status") == "complete":
        checks += 1
    if stats.get("export_status") == "complete":
        checks += 1

    for i, label in enumerate(labels):
        print(f"  {'✓' if i < checks else '✗'} {label}")

    print(f"\n  Segment count: {stats.get('segment_count', 0):,}")
    print(f"  Mean duration: {stats.get('mean_duration', 0):.2f}")
    print(f"  P95 duration: {stats.get('p95_duration', 0):.2f}")
    print(f"  Artifacts: {stats.get('artifact_count', 0)}")

    print(f"\n  Pipeline: {checks}/{total} passed")
    return checks == total


def test_btc():

    print("\n" + "=" * 60)
    print("BTC DATA VALIDATION")
    print("=" * 60)

    datasets = {
        "BTC_15m": "data/raw/BTCUSD_15m.csv",
        "BTC_1h": "data/raw/BTCUSD_1h.csv",
        "BTC_4h": "data/raw/BTCUSD_4h.csv",
    }

    missing = [
        name
        for name, path in datasets.items()
        if not Path(path).exists()
    ]

    if missing:
        print()
        print("Missing datasets:")
        print(missing)
        return False

    registry = AtlasRegistry()
    runner = ModuleRunner(registry=registry)

    results = runner.run_on_datasets(
        module_class=VolumeBehaviourResearch,
        datasets=datasets,
        output_base_dir="outputs/module017",
    )

    print()
    print("Validation")
    print("-" * 40)

    all_ok = True

    for name, result in results.items():

        print()
        print(name)

        if result.status != "completed":
            print("FAILED")
            all_ok = False
            continue

        stats = result.statistics
        artifacts = result.artifacts

        print("Completed")
        print("Segment count:", stats["segment_count"])
        print("Mean duration:", round(stats["mean_duration"], 2))
        print("P95 duration:", round(stats["p95_duration"], 2))
        print("Artifacts:", len(artifacts))

        # ---------- Occupancy ----------
        occ = artifacts["volume_occupancy"]

        if "percentage" not in occ.columns:
            print("Occupancy FAILED - missing percentage column")
            all_ok = False
        else:
            total = occ["percentage"].sum()
            print("Occupancy:", round(total, 2))

            if abs(total - 100) > 0.5:
                print("Occupancy FAILED")
                all_ok = False

        # ---------- Transition ----------
        tp = artifacts["transition_probabilities"]
        print("Transition matrix:", tp.shape)

        # ---------- Export ----------
        expected = [
            "clean_volume",
            "normalized_volume",
            "volume_percentile",
            "volume_states",
            "volume_segments",
            "volume_statistics",
            "volume_occupancy",
            "transition_counts",
            "transition_probabilities",
            "duration_distribution",
        ]

        missing_exports = [
            x
            for x in expected
            if x not in artifacts
        ]

        if missing_exports:
            print()
            print("Missing exports")
            print(missing_exports)
            all_ok = False

    return all_ok
