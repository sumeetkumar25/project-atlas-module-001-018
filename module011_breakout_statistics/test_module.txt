def create_ohlcv_deterministic(close: np.ndarray, high: np.ndarray, low: np.ndarray) -> pd.DataFrame:
    """Create OHLCV DataFrame with explicit high/low values. Deterministic and repeatable."""
    n = len(close)
    dates = pd.date_range(start="2020-01-01", periods=n, freq="15min")
    return pd.DataFrame({
        "timestamp": dates,
        "open": close,
        "high": high,
        "low": low,
        "close": close,
        "volume": np.ones(n) * 100
    })


def test_upward_breakout():
    """
    Test 1: Guaranteed upward breakout.
    
    First 25 bars oscillate inside range [95, 105].
    Bar 25: close=106, high=107 exceeds all previous highs.
    """
    print("\n" + "=" * 60)
    print("TEST 1: Upward Breakout (Deterministic)")
    print("=" * 60)
    
    n = 40
    close = np.full(n, 100.0)
    high = np.full(n, 101.0)
    low = np.full(n, 99.0)
    
    for i in range(25):
        close[i] = 100 + (i % 3) * 2
        high[i] = close[i] + 1
        low[i] = close[i] - 1
    
    close[25] = 106.0
    high[25] = 107.0
    low[25] = 105.0
    
    df = create_ohlcv_deterministic(close, high, low)
    
    research = BreakoutStatisticsResearch()
    result = research.run_dataframe(df, dataset_id="test_up_breakout")
    
    n_events = result.statistics.get("breakout_count", 0)
    up_count = result.statistics.get("up_breakouts", 0)
    
    print(f"  Max previous high (bars 5-24): {high[5:25].max():.1f}")
    print(f"  Breakout bar close: {close[25]:.1f}, high: {high[25]:.1f}")
    print(f"  Breakouts: {n_events} (up: {up_count})")
    
    checks = 0
    if n_events >= 1:
        print(f"  ✓ Breakout detected")
        checks += 1
    else:
        print(f"  ✗ No breakout detected")
    if up_count >= 1:
        print(f"  ✓ Upward breakout identified")
        checks += 1
    else:
        print(f"  ✗ Not identified as upward")
    
    print(f"  Result: {checks}/2 passed")
    return checks == 2

def test_no_breakout():
    """
    Test 3: Mathematically guaranteed no breakout.
    
    Prices oscillate [98, 100, 102]. High never exceeds 102.5, low never below 97.5.
    Close never strictly exceeds rolling high or drops below rolling low.
    """
    print("\n" + "=" * 60)
    print("TEST 3: No Breakout (Guaranteed)")
    print("=" * 60)
    
    n = 100
    
    close = np.zeros(n)
    high = np.zeros(n)
    low = np.zeros(n)
    
    for i in range(n):
        val = 98 + (i % 3) * 2
        close[i] = val
        high[i] = val + 0.5
        low[i] = val - 0.5
    
    df = create_ohlcv_deterministic(close, high, low)
    
    research = BreakoutStatisticsResearch()
    result = research.run_dataframe(df, dataset_id="test_no_breakout")
    
    n_events = result.statistics.get("breakout_count", 0)
    
    print(f"  Price range: [{low.min():.1f}, {high.max():.1f}]")
    print(f"  Breakouts: {n_events}")
    
    if n_events == 0:
        print(f"  ✓ Correctly found no breakouts")
        return True
    else:
        print(f"  ✗ Expected 0 breakouts, got {n_events}")
        return False

def test_multiple_consecutive_breakouts():
    """
    Test 5: Multiple consecutive breakouts (Option 1 - large step size).
    
    Each bar's close and high increase by 2.0.
    New close (e.g. 106) > previous highs (max was 104.5).
    This ensures every bar after warm-up is a breakout.
    
    Policy (Option A): Every qualifying candle is an independent observation.
    """
    print("\n" + "=" * 60)
    print("TEST 5: Multiple Consecutive Breakouts (Option 1)")
    print("=" * 60)
    print("  Step size: +2.0 per bar, high = close + 0.5")
    print("  Policy: Every qualifying candle = independent observation")
    
    n = 50
    lookback = 20
    
    close = np.linspace(100, 198, n)
    high = close + 0.5
    low = close - 0.5
    
    df = create_ohlcv_deterministic(close, high, low)
    
    research = BreakoutStatisticsResearch()
    result = research.run_dataframe(df, dataset_id="test_multiple")
    
    n_events = result.statistics.get("breakout_count", 0)
    up_count = result.statistics.get("up_breakouts", 0)
    
    possible_bars = n - lookback
    print(f"  Bars after warm-up: {possible_bars}")
    print(f"  Breakouts: {n_events} (up: {up_count})")
    
    checks = 0
    if n_events > 5:
        print(f"  ✓ Multiple breakouts detected ({n_events})")
        checks += 1
    else:
        print(f"  ✗ Expected multiple breakouts, got {n_events}")
    
    expected_min = possible_bars - 1
    if n_events >= expected_min:
        print(f"  ✓ Most eligible bars detected as breakouts")
        checks += 1
    else:
        print(f"  ✗ Expected at least {expected_min}, got {n_events}")
    
    print(f"  Result: {checks}/2 passed")
    return checks == 2


def test_future_leakage():
    """
    Test 6: Future leakage protection.
    
    Modify future bars and verify earlier breakout detections remain identical.
    """
    print("\n" + "=" * 60)
    print("TEST 6: Future Leakage")
    print("=" * 60)
    
    np.random.seed(42)
    n = 100
    
    close = np.cumsum(np.random.normal(0, 1, n)) + 200
    close = np.maximum(close, 1)
    high = close + np.random.uniform(0.5, 2, n)
    low = close - np.random.uniform(0.5, 2, n)
    
    df1 = create_ohlcv_deterministic(close, high, low)
    research1 = BreakoutStatisticsResearch()
    result1 = research1.run_dataframe(df1, dataset_id="test_leakage_1")
    
    close2 = close.copy()
    high2 = high.copy()
    low2 = low.copy()
    close2[-30:] = 50
    high2[-30:] = 51
    low2[-30:] = 49
    
    df2 = create_ohlcv_deterministic(close2, high2, low2)
    research2 = BreakoutStatisticsResearch()
    result2 = research2.run_dataframe(df2, dataset_id="test_leakage_2")
    
    n1 = result1.statistics.get("breakout_count", 0)
    n2 = result2.statistics.get("breakout_count", 0)
    
    print(f"  Original: {n1}, After future change: {n2}")
    
    if n1 == n2:
        print(f"  ✓ Future changes don't affect breakout count")
        return True
    else:
        print(f"  ✗ Counts differ: {n1} vs {n2}")
        return False


def test_reference_implementation():
    """
    Test 7: Reference implementation comparison.
    
    Independent implementation: close[i] > max(high[i-lookback:i]).
    """
    print("\n" + "=" * 60)
    print("TEST 7: Reference Implementation")
    print("=" * 60)
    
    np.random.seed(42)
    n = 200
    lookback = 20
    
    close = np.cumsum(np.random.normal(0, 2, n)) + 500
    close = np.maximum(close, 1)
    high = close + np.random.uniform(1, 3, n)
    low = close - np.random.uniform(1, 3, n)
    
    df = create_ohlcv_deterministic(close, high, low)
    research = BreakoutStatisticsResearch()
    result = research.run_dataframe(df, dataset_id="test_reference")
    
    ref_events = []
    for i in range(lookback, n):
        range_high = high[i-lookback:i].max()
        range_low = low[i-lookback:i].min()
        if close[i] > range_high:
            ref_events.append({"bar": i, "direction": "Up"})
        elif close[i] < range_low:
            ref_events.append({"bar": i, "direction": "Down"})
    
    atlas_count = result.statistics.get("breakout_count", 0)
    ref_count = len(ref_events)
    
    print(f"  Atlas: {atlas_count}, Reference: {ref_count}")
    
    if atlas_count == ref_count:
        print(f"  ✓ Counts match exactly")
        return True
    else:
        print(f"  ✗ Counts differ by {abs(atlas_count - ref_count)}")
        return False


def test_btc_data():
    """
    Test on all BTC datasets.
    
    Production: runs on 15m, 1h, and 4h data.
    """
    print("\n" + "=" * 60)
    print("TEST: BTC Data (All Timeframes)")
    print("=" * 60)
    
    datasets = {
        "BTC_15m": "data/raw/BTCUSD_15m.csv",
        "BTC_1h": "data/raw/BTCUSD_1h.csv",
        "BTC_4h": "data/raw/BTCUSD_4h.csv",
    }
    
    missing = [name for name, path in datasets.items() if not Path(path).exists()]
    if missing:
        print(f"  ⚠ Missing datasets: {missing}")
        print(f"  Skipping BTC integration test")
        return True
    
    registry = AtlasRegistry()
    runner = ModuleRunner(registry=registry)
    
    results = runner.run_on_datasets(
        module_class=BreakoutStatisticsResearch,
        datasets=datasets,
        output_base_dir="outputs/module011",
        validate_after=False
    )
    
    all_passed = True
    for name, result in results.items():
        if result.status == "completed":
            n = result.statistics.get("total_breakouts", 0)
            up = result.statistics.get("up_breakouts", 0)
            down = result.statistics.get("down_breakouts", 0)
            mfe = result.statistics.get("mean_mfe_pct", 0)
            mae = result.statistics.get("mean_mae_pct", 0)
            fail = result.statistics.get("failure_rate", 0)
            
            print(f"\n  {name}:")
            print(f"    Breakouts: {n:,} ({up} up, {down} down)")
            print(f"    Mean MFE: {mfe:.4f}%")
            print(f"    Mean MAE: {mae:.4f}%")
            print(f"    Failure rate: {fail*100:.1f}%")
            print(f"    ✓ Analysis complete")
        else:
            print(f"  ✗ {name}: Failed - {result.errors}")
            all_passed = False
    
    return all_passed
