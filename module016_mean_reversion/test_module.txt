def test_pipeline_execution():
    """Verify all 7 steps execute correctly."""
    print("\n" + "=" * 60)
    print("TEST: Full Pipeline (All 7 Steps)")
    print("=" * 60)
    
    n = 500
    np.random.seed(42)
    close = np.cumsum(np.random.normal(0, 50, n)) + 50000
    close = np.maximum(close, 1)
    
    dates = pd.date_range(start="2020-01-01", periods=n, freq="15min")
    df = pd.DataFrame({
        "timestamp": dates,
        "open": close * 0.999,
        "high": close * 1.005,
        "low": close * 0.995,
        "close": close,
        "volume": np.ones(n) * 100
    })
    
    research = MeanReversionResearch()
    result = research.run_dataframe(df, dataset_id="phase1_6_final_test")
    
    stats = result.statistics
    checks = 0
    
    labels = [
        "Completed",
        "Deviation",
        "Extremes",
        "Reversion",
        "Duration",
        "Magnitude",
        "Summary",
        "Export",
        "Reverted > 0",
        "Mean duration > 0",
        "Mean recovery > 0",
        "Artifact count > 0",
        "Research summary",
        "Distribution",
        "Reverted ≤ Total",
        "Min recovery ≥ 0",
        "Max recovery ≤ 1.0",
    ]
    
    total = len(labels)
    
    if result.status == "completed": checks += 1
    if stats.get("deviation_status") == "complete": checks += 1
    if stats.get("extremes_status") == "complete": checks += 1
    if stats.get("reversion_status") == "complete": checks += 1
    if stats.get("duration_status") == "complete": checks += 1
    if stats.get("magnitude_status") == "complete": checks += 1
    if stats.get("summary_status") == "complete": checks += 1
    if stats.get("export_status") == "complete": checks += 1
    if stats.get("reverted_events", 0) > 0: checks += 1
    if stats.get("mean_duration", 0) > 0: checks += 1
    if stats.get("mean_recovery_ratio", 0) > 0: checks += 1
    if stats.get("artifact_count", 0) > 0: checks += 1
    if "research_summary" in result.artifacts: checks += 1
    if "reversion_magnitude_distribution" in result.artifacts: checks += 1
    if stats.get("reverted_events", 0) <= stats.get("total_extremes", 1): checks += 1
    
    if stats.get("reverted_events", 0) > 0:
        if stats.get("min_recovery_ratio", -1) >= 0: checks += 1
        if stats.get("max_recovery_ratio", 2) <= 1.0: checks += 1
    
    for i, label in enumerate(labels):
        print(f"  {'✓' if i < checks else '✗'} {label}")
    
    reverted = stats.get("reverted_events", 0)
    total_ext = stats.get("total_extremes", 0)
    mean_rec = stats.get("mean_recovery_ratio", 0)
    artifact_count = stats.get("artifact_count", 0)
    
    print(f"\n  Extreme events: {total_ext:,}")
    print(f"  Reverted: {reverted:,}")
    print(f"  Mean recovery: {mean_rec:.2%}")
    print(f"  Artifacts exported: {artifact_count}")
    
    print(f"\n  Pipeline: {checks}/{total} passed")
    return checks == total


def test_btc_data():
    """Test on BTC data."""
    print("\n" + "=" * 60)
    print("TEST: BTC Data (15m)")
    print("=" * 60)
    
    datasets = {"BTC_15m": "data/raw/BTCUSD_15m.csv"}
    
    missing = [name for name, path in datasets.items() if not Path(path).exists()]
    if missing:
        print(f"  ⚠ Missing: {missing}")
        return True
    
    registry = AtlasRegistry()
    runner = ModuleRunner(registry=registry)
    
    results = runner.run_on_datasets(
        module_class=MeanReversionResearch,
        datasets=datasets,
        output_base_dir="outputs/module016",
        validate_after=False
    )
    
    for name, result in results.items():
        if result.status == "completed":
            stats = result.statistics
            total_ext = stats.get("total_extremes", 0)
            reverted = stats.get("reverted_events", 0)
            mean_dur = stats.get("mean_duration", 0)
            mean_rec = stats.get("mean_recovery_ratio", 0)
            artifact_count = stats.get("artifact_count", 0)
            
            print(f"\n  {name}:")
            print(f"    Extreme events: {total_ext:,}")
            print(f"    Reverted: {reverted:,}")
            print(f"    Mean duration: {mean_dur:.2f} bars")
            print(f"    Mean recovery: {mean_rec:.2%}")
            print(f"    Artifacts: {artifact_count}")
            
            checks = 0
            if stats.get("deviation_status") == "complete": checks += 1
            if stats.get("extremes_status") == "complete": checks += 1
            if stats.get("reversion_status") == "complete": checks += 1
            if stats.get("duration_status") == "complete": checks += 1
            if stats.get("magnitude_status") == "complete": checks += 1
            if stats.get("summary_status") == "complete": checks += 1
            if stats.get("export_status") == "complete": checks += 1
            if reverted > 0: checks += 1
            if artifact_count > 0: checks += 1
            
            print(f"    Checks: {checks}/9 passed")
            return checks == 9
    return False
