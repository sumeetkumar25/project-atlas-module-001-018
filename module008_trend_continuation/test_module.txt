def create_price_dataframe(prices: np.ndarray) -> pd.DataFrame:
    """Create OHLCV DataFrame from price series"""
    n = len(prices)
    dates = pd.date_range(start="2020-01-01", periods=n, freq="15min")
    
    high = np.maximum(prices, np.roll(prices, -1))
    low = np.minimum(prices, np.roll(prices, -1))
    high[-1] = prices[-1] * 1.001
    low[-1] = prices[-1] * 0.999
    
    return pd.DataFrame({
        "timestamp": dates,
        "open": prices,
        "high": high,
        "low": low,
        "close": prices,
        "volume": np.ones(n) * 100
    })


def run_full_chain(df: pd.DataFrame, use_adaptive: bool = False) -> dict:
    """
    Run the complete chain: Module 005 → Module 007 → Module 008
    All in-memory - no disk I/O.
    """
    # Module 005
    trend_module = TrendPersistenceResearch()
    trend_module.MIN_TREND_LENGTH = 1
    trend_result = trend_module.run_dataframe(df, dataset_id="chain_trends")
    
    if 'trend_segments' not in trend_result.artifacts:
        return {"error": "Module 005: no trends"}
    
    trends = trend_result.artifacts['trend_segments']
    
    # Module 007
    pullback_module = PullbackResearch()
    pullback_module.set_trend_segments(trends)
    pullback_result = pullback_module.run_dataframe(df, dataset_id="chain_pullbacks")
    
    if 'pullbacks' not in pullback_result.artifacts:
        return {"error": "Module 007: no pullbacks"}
    
    pullbacks = pullback_result.artifacts['pullbacks']
    
    # Module 008
    continuation_module = TrendContinuationResearch()
    if not use_adaptive:
        # Override to fixed threshold for controlled testing
        continuation_module._get_threshold_pct = lambda: 0.50
    
    continuation_module.set_pullbacks(pullbacks)
    result = continuation_module.run_dataframe(df, dataset_id="chain_continuation")
    
    return {
        "trends": trends,
        "pullbacks": pullbacks,
        "events": result.artifacts.get('continuation_events'),
        "result": result
    }


def test_synthetic_continuation():
    """
    Test 1: Price clearly continues after pullback.
    
    Pattern: uptrend → small pullback → strong continuation
    The pullback ends at a low, then price rises significantly.
    """
    print("\n" + "=" * 60)
    print("SYNTHETIC TEST 1: Clear Continuation")
    print("=" * 60)
    
    # Build price series with clear continuation after pullback
    prices = []
    # Steady uptrend (20 bars)
    prices.extend(np.linspace(100, 120, 20))
    # Small pullback (5 bars, -4%)
    prices.extend([120, 118, 117, 116, 115.5])
    # Strong continuation (30 bars, +25%)
    prices.extend(np.linspace(115.5, 145, 30))
    
    prices = np.array(prices)
    df = create_price_dataframe(prices)
    
    print(f"\n  Price path: {len(prices)} bars")
    print(f"  Pattern: 100→120 uptrend → pullback to 115.5 → continuation to 145")
    
    chain = run_full_chain(df, use_adaptive=False)
    
    if "error" in chain:
        print(f"  ⚠ {chain['error']}")
        return False
    
    events = chain['events']
    result = chain['result']
    
    if events is None or len(events) == 0:
        print(f"  ⚠ No events - chain produced no pullbacks")
        return True
    
    n_cont = int((events['outcome'] == 'continuation').sum())
    n_rev = int((events['outcome'] == 'reversal').sum())
    n_total = len(events)
    
    print(f"\n  Chain: {len(chain['trends'])} trends → {len(chain['pullbacks'])} pullbacks → {n_total} events")
    print(f"  Continuation: {n_cont}/{n_total} ({n_cont/n_total*100:.0f}%)")
    print(f"  Reversal: {n_rev}/{n_total} ({n_rev/n_total*100:.0f}%)")
    
    # Show each event
    print(f"\n  Events:")
    for _, ev in events.iterrows():
        print(f"    Trend {int(ev['trend_id'])}: {ev['trend_direction']}, "
              f"pb_depth={ev['pullback_depth_pct']:.2f}%, "
              f"outcome={ev['outcome']}, "
              f"pct_change={ev['pct_change_from_pb_end']:+.2f}%")
    
    # This is a continuation scenario - expect majority continuation
    if n_cont > n_rev:
        print(f"\n  ✓ Continuation dominates (expected for this pattern)")
        return True
    elif n_cont > 0:
        print(f"\n  ○ Some continuation detected")
        return True
    else:
        print(f"\n  ✗ No continuation detected in clear continuation scenario")
        return False

def test_cross_module_consistency():
    """
    Test 3: Cross-validate Module 007 recovery with Module 008 continuation.
    
    Pullbacks that Module 007 flags as "recovered" should show
    high continuation rates in Module 008.
    """
    print("\n" + "=" * 60)
    print("SYNTHETIC TEST 3: Cross-Module Consistency")
    print("=" * 60)
    
    # Create price path where recovery and continuation should align
    prices = []
    prices.extend(np.linspace(100, 130, 30))    # Strong uptrend
    prices.extend([130, 127, 125, 124, 123])     # Pullback
    prices.extend(np.linspace(123, 140, 20))     # Recovery + continuation
    
    prices = np.array(prices)
    df = create_price_dataframe(prices)
    
    chain = run_full_chain(df, use_adaptive=False)
    
    if "error" in chain:
        print(f"  ⚠ {chain['error']}")
        return False
    
    pullbacks = chain['pullbacks']
    events = chain['events']
    
    if pullbacks is None or events is None:
        print(f"  ⚠ No data")
        return True
    
    # Check Module 007 recovery vs Module 008 continuation
    print(f"\n  Pullback → Recovery → Continuation mapping:")
    
    consistent = 0
    inconsistent = 0
    
    for _, pb in pullbacks.iterrows():
        pb_id = pb['trend_id']
        recovered = pb.get('recovered', False)
        
        # Find matching event
        matching = events[events['trend_id'] == pb_id]
        if len(matching) > 0:
            outcome = matching.iloc[0]['outcome']
            
            if recovered and outcome == 'continuation':
                consistent += 1
                print(f"    ✓ Trend {int(pb_id)}: recovered + continuation")
            elif recovered and outcome != 'continuation':
                inconsistent += 1
                print(f"    ⚠ Trend {int(pb_id)}: recovered but {outcome}")
            elif not recovered and outcome == 'reversal':
                consistent += 1
                print(f"    ✓ Trend {int(pb_id)}: not recovered + reversal")
    
    print(f"\n  Consistent: {consistent}, Inconsistent: {inconsistent}")
    
    if consistent >= inconsistent:
        print(f"  ✓ Modules 007 and 008 are reasonably consistent")
    else:
        print(f"  ⚠ Modules may be measuring different concepts")
    
    return True


def test_btc_datasets():
    """Test on BTC real data with adaptive thresholds"""
    print("\n" + "=" * 60)
    print("BTC DATASETS (Adaptive ATR Thresholds)")
    print("=" * 60)
    
    registry = AtlasRegistry()
    runner = ModuleRunner(registry=registry)
    
    datasets = {
        "BTC_15m": "data/raw/BTCUSD_15m.csv",
        "BTC_1h": "data/raw/BTCUSD_1h.csv",
        "BTC_4h": "data/raw/BTCUSD_4h.csv",
    }
    
    missing = [name for name, path in datasets.items() if not Path(path).exists()]
    if missing:
        print(f"\n✗ Missing: {missing}")
        return False
    
    results = runner.run_on_datasets(
        module_class=TrendContinuationResearch,
        datasets=datasets,
        output_base_dir="outputs/module008",
        validate_after=False
    )
    
    all_passed = True
    
    for name, result in results.items():
        if result.status == "completed":
            cont_rate = result.statistics.get("continuation_rate", 0)
            rev_rate = result.statistics.get("reversal_rate", 0)
            und_rate = result.statistics.get("undecided_rate", 0)
            n_events = result.statistics.get("events_evaluated", 0)
            threshold = result.statistics.get("threshold_pct", 0)
            
            print(f"\n  {name} (threshold={threshold:.4f}%):")
            print(f"    Events: {n_events:,}")
            print(f"    Continuation: {cont_rate*100:.1f}%")
            print(f"    Reversal: {rev_rate*100:.1f}%")
            print(f"    Undecided: {und_rate*100:.1f}%")
            
            if 'depth_probability' in result.artifacts:
                depth_table = result.artifacts['depth_probability']
                print(f"\n    By depth:")
                for _, row in depth_table.iterrows():
                    print(f"      {row['depth_bin']:>10}: {row['continuation_pct']:5.1f}% cont "
                          f"(n={int(row['sample_size']):,})")
            
            if n_events > 0:
                print(f"    ✓ Analysis complete")
            else:
                print(f"    ⚠ No events")
        else:
            print(f"  ✗ {name}: Failed")
            all_passed = False
    
    return all_passed
