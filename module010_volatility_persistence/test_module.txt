def test_pipeline_all_three():
    """Test pipeline with returns + volatility + percentiles."""
    print("\n" + "=" * 60)
    print("TEST: Pipeline (Returns + Vol + Percentiles)")
    print("=" * 60)
    
    n = 2000
    np.random.seed(42)
    prices = 50000 + np.cumsum(np.random.normal(0, 100, n))
    prices = np.maximum(prices, 1)
    
    dates = pd.date_range(start="2020-01-01", periods=n, freq="15min")
    df = pd.DataFrame({
        "timestamp": dates, "open": prices*0.999, "high": prices*1.002,
        "low": prices*0.998, "close": prices, "volume": np.ones(n)*100
    })
    
    research = VolatilityPersistenceResearch()
    result = research.run_dataframe(df, dataset_id="phase2_3_test")
    
    checks = 0
    labels = [
        "Module completed",
        "Returns calculated",
        "Volatility calculated",
        "Percentiles calculated",
        "Return warm-up = 1",
        "Volatility warm-up = 20",
        "Percentile warm-up > 0",
        "Valid percentiles > 0",
        "Percentile CSV exported",
    ]
    
    if result.status == "completed":
        checks += 1
    if result.statistics.get("returns_status") == "complete":
        checks += 1
    if result.statistics.get("volatility_status") == "complete":
        checks += 1
    if result.statistics.get("percentiles_status") == "complete":
        checks += 1
    if result.statistics.get("return_warmup_nan") == 1:
        checks += 1
    if result.statistics.get("volatility_warmup_nan") == 20:
        checks += 1
    if result.statistics.get("percentile_warmup_nan", 0) > 0:
        checks += 1
    if result.statistics.get("valid_percentiles", 0) > 0:
        checks += 1
    if 'percentiles' in result.artifacts:
        checks += 1
    
    for i, label in enumerate(labels):
        passed = i < checks
        print(f"  {'✓' if passed else '✗'} {label}")
    
    # Show warm-up details
    ret_warm = result.statistics.get("return_warmup_nan", "?")
    vol_warm = result.statistics.get("volatility_warmup_nan", "?")
    pct_warm = result.statistics.get("percentile_warmup_nan", "?")
    valid_pct = result.statistics.get("valid_percentiles", "?")
    pct_mean = result.statistics.get("percentile_mean", 0)
    
    print(f"\n  Warm-up summary:")
    print(f"    Returns:    {ret_warm} NaN")
    print(f"    Volatility: {vol_warm} NaN")
    print(f"    Percentile: {pct_warm} NaN")
    print(f"    Valid pct:  {valid_pct}")
    print(f"    Mean pct:   {pct_mean:.2f}" if isinstance(pct_mean, (int, float)) else f"    Mean pct:   {pct_mean}")
    
    print(f"\n  Pipeline: {checks}/{len(labels)} passed")
    return checks >= len(labels) - 2


def test_btc_data():
    """Test on BTC data."""
    print("\n" + "=" * 60)
    print("TEST: BTC Data (15m)")
    print("=" * 60)
    
    datasets = {"BTC_15m": "data/raw/BTCUSD_15m.csv"}
    
    missing = [name for name, path in datasets.items() if not Path(path).exists()]
    if missing:
        print(f"  ⚠ Missing: {missing}")
        return True
    
    registry = AtlasRegistry()
    runner = ModuleRunner(registry=registry)
    
    results = runner.run_on_datasets(
        module_class=VolatilityPersistenceResearch,
        datasets=datasets,
        output_base_dir="outputs/module010",
        validate_after=False
    )
    
    for name, result in results.items():
        if result.status == "completed":
            nprices = result.statistics.get("price_observations", 0)
            nret = result.statistics.get("return_observations", 0)
            ret_warm = result.statistics.get("return_warmup_nan", -1)
            vol_warm = result.statistics.get("volatility_warmup_nan", -1)
            pct_warm = result.statistics.get("percentile_warmup_nan", -1)
            nvol = result.statistics.get("valid_volatility", 0)
            npct = result.statistics.get("valid_percentiles", 0)
            pct_mean = result.statistics.get("percentile_mean", 0)
            
            print(f"\n  {name}:")
            print(f"    Prices:     {nprices:,}")
            print(f"    Returns:    {nret:,} (warm-up: {ret_warm})")
            print(f"    Volatility: valid={nvol:,}, warm-up={vol_warm}")
            print(f"    Percentile: valid={npct:,}, warm-up={pct_warm}")
            print(f"    Mean pct:   {pct_mean:.2f}" if isinstance(pct_mean, (int, float)) else f"    Mean pct:   {pct_mean}")
            
            checks = sum([
                ret_warm == 1,
                vol_warm == 20,
                pct_warm > 0,
                nvol > 0,
                npct > 0,
                nret == nprices,
            ])
            print(f"    {'✓' if checks >= 5 else '⚠'} {checks}/6 checks passed")
            return checks >= 5
    return False
